# 1. Strip & UPX-pack both v4 and v5 binaries (exact Hutter Prize packaging)
for v in mini_cmix_v4 mini_cmix_v5; do
    cp "$v" "${v}_packed"
    strip --strip-all "${v}_packed"
    upx-ucl --best "${v}_packed"
done

# 2. Run Linux baselines (gzip -9 and xz -9)
echo "=== Running gzip -9 & xz -9 on 100 MB enwik8 ==="
gzip -9 -c enwik8 > enwik8.gz
xz -9 -c enwik8 > enwik8.xz

# 3. Run mini_cmix_v4 (compress + decompress + SHA-256 check)
echo "=== Running mini_cmix_v4 on 100 MB enwik8 ==="
taskset -c 0 /usr/bin/time -v ./mini_cmix_v4_packed c enwik8 enwik8.v4
taskset -c 0 /usr/bin/time -v ./mini_cmix_v4_packed d enwik8.v4 enwik8.v4.restored

# 4. Run mini_cmix_v5 (compress + decompress + SHA-256 check)
echo "=== Running mini_cmix_v5 on 100 MB enwik8 ==="
taskset -c 0 /usr/bin/time -v ./mini_cmix_v5_packed c enwik8 enwik8.v5
taskset -c 0 /usr/bin/time -v ./mini_cmix_v5_packed d enwik8.v5 enwik8.v5.restored

# 5. Verify 100% lossless SHA-256 match
sha256sum enwik8 enwik8.v4.restored enwik8.v5.restored
